#if UNITY_EDITOR
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEngine;
using UnityEngine.Rendering;

namespace RVFX.PlaybackSupport
{
    // No reference to RVFX DLLs: diagnoses Content Only imports even if runtime is absent.
    [InitializeOnLoad]
    internal sealed class RVFXPlaybackPackageCheck : IPreprocessBuildWithReport
    {
        [Serializable] private sealed class Requirement { public string name, version; }
        [Serializable] private sealed class Manifest
        {
            public int contentContract;
            public string unityVersion, pipeline;
            public string[] prefabs, prefabGuids;
            public Requirement[] packages;
        }
        public int callbackOrder => -1000;
        static RVFXPlaybackPackageCheck() { EditorApplication.delayCall += CheckAfterImport; }
        private static void CheckAfterImport() { Validate(false); }

        [MenuItem("Tools/RVFX/Playback/Validate Imported Packages")]
        public static void ValidateMenu() { Validate(true); }
        public void OnPreprocessBuild(BuildReport report)
        {
            var errors = Validate(false);
            if (errors.Count > 0) throw new BuildFailedException("RVFX playback package validation:\n" + string.Join("\n", errors));
        }

        public static List<string> Validate(bool verbose)
        {
            var errors = new List<string>();
            var type = AppDomain.CurrentDomain.GetAssemblies().Select(a => a.GetType("RVFX.DestructionSystem.RVFXPlaybackRuntimeInfo")).FirstOrDefault(t => t != null);
            int contract = type == null ? 0 : (int)type.GetField("ContentContract").GetRawConstantValue();
            var packages = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages();
            string pipeline = GraphicsSettings.currentRenderPipeline == null ? "BuiltIn" :
                GraphicsSettings.currentRenderPipeline.GetType().Name.Contains("Universal") ? "URP" :
                GraphicsSettings.currentRenderPipeline.GetType().Name.Contains("HDRender") ? "HDRP" : "Other";
            foreach (string file in AssetDatabase.FindAssets("RVFXPlaybackPackage_ t:TextAsset").Select(AssetDatabase.GUIDToAssetPath))
            {
                var m = JsonUtility.FromJson<Manifest>(File.ReadAllText(file));
                if (m == null || m.prefabs == null) { errors.Add("Invalid playback manifest: " + file); continue; }
                if (contract < m.contentContract) errors.Add("Playback Runtime is missing or older than content contract " + m.contentContract + ". Import Include Playback Runtime from the authoring project.");
                if (m.pipeline != pipeline) errors.Add("Playback package requires " + m.pipeline + "; active pipeline is " + pipeline + ". Configure the matching pipeline or repackage converted content.");
                foreach (var r in m.packages ?? new Requirement[0])
                {
                    var installed = packages.FirstOrDefault(p => p.name == r.name);
                    if (installed == null) errors.Add("Install " + r.name + " using Package Manager. Source version: " + r.version);
                    else if (verbose && installed.version != r.version) Debug.Log("RVFX: " + r.name + " " + installed.version + " installed (source " + r.version + "). A different version is not automatically a failure; verify rendering and playback.");
                }
                for (int i = 0; i < m.prefabs.Length; i++)
                {
                    string path = m.prefabGuids != null && i < m.prefabGuids.Length ? AssetDatabase.GUIDToAssetPath(m.prefabGuids[i]) : m.prefabs[i];
                    var prefab = AssetDatabase.LoadAssetAtPath<GameObject>(path);
                    if (prefab == null) { errors.Add("Missing packaged prefab: " + path); continue; }
                    foreach (var t in prefab.GetComponentsInChildren<Transform>(true))
                    {
                        if (GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject) != 0) errors.Add("Missing playback/effect script: " + path + "/" + t.name);
                        foreach (var c in t.GetComponents<Component>())
                        {
                            if (c == null) continue;
                            using (var so = new SerializedObject(c))
                            {
                                var p = so.GetIterator();
                                while (p.Next(true))
                                    if (p.propertyType == SerializedPropertyType.ObjectReference && p.objectReferenceValue == null && p.objectReferenceInstanceIDValue != 0)
                                        errors.Add("Missing reference: " + path + " / " + p.propertyPath);
                            }
                        }
                    }
                }
            }
            foreach (string error in errors.Distinct()) Debug.LogWarning("RVFX: " + error);
            if (verbose && errors.Count == 0) Debug.Log("RVFX playback package references and runtime contract are valid. Test actual playback and rendering on your target platform.");
            return errors.Distinct().ToList();
        }
    }
}
#endif
